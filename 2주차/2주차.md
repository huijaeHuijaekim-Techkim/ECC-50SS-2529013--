**# 1. ORDER BY 복수 열 정렬 및 LIMIT 행 제한**



\-- 테스트 DB 선택 및 샘플 테이블(sample32) 생성

USE testdb;



CREATE TABLE sample32 (

&#x20;   a INT,

&#x20;   b INT

);



\-- 데이터 삽입

INSERT INTO sample32 (a, b) VALUES

(1, 1),

(2, 1),

(2, 2),

(1, 3),

(1, 2);



\-- 복수 열 정렬: 

\-- a열을 오름차순으로 먼저 정렬한 후, a값이 동일한 행들에 한하여 b열을 오름차순 정렬함.

SELECT \* FROM sample32 ORDER BY a, b;





**# 2. 사칙연산과 ALIAS**

\-- 1. AS 키워드로 한글 별명 부여하기 (한글 객체명은 큰따옴표 "" 사용)

SELECT price, quantity, price \* quantity AS "총금액" FROM sample34;



\-- 2. WHERE 절에서는 SELECT에서 만든 별명을 사용할 수 없음 (에러 발생 코드)

\-- SELECT \*, price \* quantity AS amount FROM sample34 WHERE amount >= 2000;



\-- 3. 올바른 조건식 작성 및 ORDER BY에서의 별명 활용

\-- ORDER BY는 가장 마지막에 실행되므로 별명(amount)을 문제없이 인식함.

SELECT \*, price \* quantity AS amount 

&#x20; FROM sample34 

&#x20;WHERE price \* quantity >= 2000 

&#x20;ORDER BY amount DESC;





**4장\_데이터 추가, 삭제, 갱신**

**# 1. INSERT (데이터 추가)**

\-- 1. 기본 데이터 추가 (모든 열에 값 넣기)

\-- 테이블 구조에 맞게 순서대로 값을 입력함

INSERT INTO users (id, name, age) VALUES (1, '홍길동', 25);



\-- 2. 특정 열만 지정하여 데이터 추가

\-- age 등 생략된 열이 있다면 NULL이 들어가거나, 설정된 DEFAULT 값으로 자동 채워짐

INSERT INTO users (id, name) VALUES (2, '김철수');



\-- 3. (에러 발생 예시) NOT NULL 열에 NULL 삽입 시도

\-- id 열이 NOT NULL 제약조건을 가질 경우 아래 코드는 실행 실패함

INSERT INTO users (id, name) VALUES (NULL, '이영희');



**# 2. UPDATE (데이터 갱신)**

\-- 1. 단일 열 갱신

\-- id가 1인 사용자의 나이를 26으로 수정

UPDATE users SET age = 26 WHERE id = 1;



\-- 2. 복수 열 동시 갱신

\-- 쉼표(,)를 사용하여 여러 값을 한 번에 변경

UPDATE users SET name = '홍길동(수정)', age = 27 WHERE id = 1;



\-- 3. 증가 연산 활용

\-- 특정 조건 없이 실행할 경우, 모든 데이터의 no 열 값이 1씩 증가함

UPDATE sample41 SET no = no + 1;



\-- 4. NULL로 데이터 갱신 (데이터 초기화)

\-- 특정 행의 데이터를 지워 빈칸(NULL)으로 만듦

UPDATE sample41 SET a = NULL WHERE id = 5;





**# 3. DELETE (데이터 삭제)** 

\-- 1. 물리 삭제 (DELETE)

\-- WHERE 조건에 해당하는 실제 데이터를 DB에서 영구 삭제

DELETE FROM users WHERE id = 2;



\-- \[참고] 논리 삭제 패턴 (UPDATE 활용)

\-- 데이터를 실제로 지우지 않고, 'is\_deleted' 등의 삭제 플래그 열을 1(true)로 갱신함

UPDATE users SET is\_deleted = 1 WHERE id = 3;



\-- 논리 삭제된 데이터를 제외하고 정상 데이터만 조회하는 SELECT 문

SELECT \* FROM users WHERE is\_deleted = 0;

**# 1. 원격 저장소 복제 (Clone)**

\# 원격 저장소를 현재 디렉터리로 복제 (HTTPS 주소 사용)

git clone https://github.com/아이디/저장소명.git .



**# 2. 다중 환경(협업)에서의 동기화 (Pull \& Push)**

\# 작업 시작 전: 원격 저장소의 최신 커밋 내역을 내 컴퓨터로 가져와 병합함

git pull origin main



\# 작업 완료 후: 내 컴퓨터에서 만든 새로운 커밋을 원격 저장소로 밀어 올림

git push origin main





**# 3. 원격 브랜치 정보 페치(Fetch)와 병합 검토**

\# 1. 원격 저장소의 최신 변경 '정보만' 가져오기 (병합은 안 함)

git fetch



\# 2. 페치해 온 원격의 최신 커밋(origin/main)과 내 지역 저장소의 최신 커밋(HEAD) 차이 비교

git diff HEAD origin/main



\# 3. 차이를 검토한 결과 문제가 없다면 수동으로 병합 처리

git merge origin/main





**# 4. 협업을 위한 브랜치 작업 프로세스**

\# 1. 팀원 각자의 작업을 위한 새로운 브랜치 생성 및 이동

git switch -c plan



\# 2. 작업 완료 후 파일 스테이징 및 커밋

git add .

git commit -m "기획 문서 초안 작성"



\# 3. 원격 저장소에 나의 브랜치 이름(plan)으로 최초 푸시 

\# (이후 깃허브 웹사이트에 접속하여 Pull Request 버튼을 눌러 병합을 요청함)

git push -u origin plan
