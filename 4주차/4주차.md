

**7장\_복수\_테이블**



\-- 1. 집합 연산: UNION (합집합, 중복 제거)

\-- 열 이름이 서로 다를 경우 AS를 사용해 이름을 맞추고 마지막에 정렬(ORDER BY) 적용

SELECT a AS c FROM sample71\_a

UNION

SELECT b AS c FROM sample71\_b ORDER BY c;



\-- 2. 집합 연산: UNION ALL (중복값 유지)

\-- 중복 제거 연산을 수행하지 않음

SELECT \* FROM sample71\_a

UNION ALL

SELECT \* FROM sample71\_b;



\-- 3. 집합 연산: INTERSECT (교집합) / EXCEPT (차집합)

SELECT \* FROM sample71\_a INTERSECT SELECT \* FROM sample71\_b;

SELECT \* FROM sample71\_a EXCEPT SELECT \* FROM sample71\_b;

\-- 결합(Join) 실습을 위한 DDL (테이블 준비)

\-- =========================================================

CREATE TABLE 상품 (

&#x20;   상품코드 CHAR(4) NOT NULL,

&#x20;   상품명 VARCHAR(30),

&#x20;   메이커명 VARCHAR(30),

&#x20;   가격 INTEGER,

&#x20;   상품분류 VARCHAR(30),

&#x20;   PRIMARY KEY(상품코드)

);



CREATE TABLE 재고수 (

&#x20;   상품코드 CHAR(4),

&#x20;   입고날짜 DATE,

&#x20;   재고수 INTEGER

);





\-- =========================================================

\-- 테이블 결합 (CROSS, INNER, OUTER, SELF) 



\-- 4. 교차 결합 (Cross Join)

\-- 조건 없이 두 테이블의 모든 행을 조합한 결과를 반환함 (곱집합)

SELECT \* FROM 상품, 재고수;

\-- SELECT \* FROM 상품 CROSS JOIN 재고수;



\-- 5. WHERE 조건을 활용한 내부 결합

\-- 교차 결합 후 WHERE 구문으로 특정 조건(상품코드가 일치함)을 걸어 필터링

SELECT \* FROM 상품, 재고수

&#x20;WHERE 상품.상품코드 = 재고수.상품코드;



\-- 검색할 행과 반환할 열 지정 

SELECT 상품.상품명, 재고수.재고수 

&#x20; FROM 상품, 재고수

&#x20;WHERE 상품.상품코드 = 재고수.상품코드

&#x20;  AND 상품.상품분류 = '식료품';



\-- 6. INNER JOIN을 활용한 내부 결합

\-- 양쪽 테이블에 모두 존재하는 데이터만 결과에 포함

SELECT 상품.상품명, 재고수.재고수 

&#x20; FROM 상품 INNER JOIN 재고수 

&#x20;   ON 상품.상품코드 = 재고수.상품코드;



\-- 7. 자가 결합 (Self Join)

\-- 물리적으로는 1개의 테이블이지만 ALIAS(별명)를 통해 2개인 것처럼 엮음

SELECT S1.상품명 AS 상품A, S2.상품명 AS 상품B

&#x20; FROM 상품 S1 INNER JOIN 상품 S2

&#x20;   ON S1.상품코드 = S2.상품코드;



\-- 8. 외부 결합 (LEFT JOIN)

\-- 기준이 되는 테이블(상품3)의 데이터는 조건과 일치하지 않아도 모두 보존

SELECT 상품3.상품명, 재고수.재고수

&#x20; FROM 상품3 LEFT JOIN 재고수

&#x20;   ON 상품3.상품코드 = 재고수.상품코드

&#x20;WHERE 상품3.상품분류 = '식료품';





**8장\_데이터베이스\_설계**



**#1.AUTO\_INCREMENT를 활용한 테이블 설계**

\-- 1. 제품 관리를 위한 카탈로그 테이블 (LOB 활용)

CREATE TABLE 상품카탈로그 (

&#x20;   -- 고정 길이 문자열로 상품코드 지정 (제1 정규형 고려)

&#x20;   상품코드 CHAR(4) PRIMARY KEY, 

&#x20;   -- 큰 변동폭을 가지는 데이터는 가변 길이 문자열 활용

&#x20;   상품설명 VARCHAR(1000), 

&#x20;   -- 이미지 등 수천 바이트의 거대 데이터는 LOB 형 사용 (인덱스 불가)

&#x20;   상품이미지 BLOB

);



\-- 2. 주문 번호 자동 증가 테이블

CREATE TABLE 주문 (

&#x20;   -- INSERT 시 값을 넣지 않아도 1, 2, 3... 순서대로 자동 배정됨

&#x20;   주문번호 INTEGER AUTO\_INCREMENT,

&#x20;   주문일자 DATE,

&#x20;   고객번호 INTEGER,

&#x20;   -- AUTO\_INCREMENT 열은 반드시 PRIMARY KEY 또는 UNIQUE 제약이 필요함

&#x20;   PRIMARY KEY (주문번호) 

);



\-- 주문 상품 상세 테이블 (제2 정규형 고려: 주문 테이블과 분할 연계)

CREATE TABLE 주문상품 (

&#x20;   주문번호 INTEGER,

&#x20;   상품코드 CHAR(4),

&#x20;   수량 INTEGER,

&#x20;   -- 복합 기본키 설정 (주문번호 + 상품코드)

&#x20;   PRIMARY KEY (주문번호, 상품코드)

);
