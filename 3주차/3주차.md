**# 1. 집계 함수와 그룹화 (GROUP BY, HAVING)**



1\. 행 개수 구하기 및 중복 제거

SELECT COUNT(\*) FROM sample51; -- NULL 값을 포함한 전체 행 수 반환

SELECT COUNT(no) FROM sample51; -- no 열이 NULL인 행은 무시하고 반환

SELECT DISTINCT name FROM sample51; -- 중복을 제거한 name 반환

SELECT COUNT(DISTINCT name) FROM sample51; -- 중복을 제거한 후의 고유한 name 개수 반환



2\. CASE 문을 활용한 NULL 대체 및 평균 구하기

SELECT AVG(CASE WHEN quantity IS NULL THEN 0 ELSE quantity END) AS avgnull0 

&#x20; FROM sample51;



3\. 그룹화 및 HAVING 조건 필터링

\-- name별로 그룹을 묶고, 그룹 내 개수가 1개인 데이터만 내림차순 정렬

SELECT name, COUNT(name), SUM(quantity)

&#x20; FROM sample51

&#x20;GROUP BY name

HAVING COUNT(name) = 1

&#x20;ORDER BY SUM(quantity) DESC;





**# 2. 서브쿼리(Subquery)와 상관 서브쿼리**



1\. 스칼라 서브쿼리를 활용한 데이터 삭제 (a값이 가장 작은 행 삭제)

&#x20;WHERE a = (SELECT MIN(a) FROM sample54);



2\. INSERT 명령에서 서브쿼리 활용

INSERT INTO sample541 SELECT 1, 2;

INSERT INTO sample542 SELECT \* FROM sample543;



3\. 상관 서브쿼리 (EXISTS 활용)

UPDATE sample551 

&#x20;  SET a = '있음' 

&#x20;WHERE EXISTS (SELECT \* 

&#x20;                FROM sample552 

&#x20;               WHERE sample552.no\_2 = sample551.no);



4\. IN 집합을 활용한 조건식

SELECT \* FROM sample551 

&#x20;WHERE no IN (SELECT no\_2 FROM sample552);



**3. DDL: 데이터베이스 객체 작성과 삭제**

1\. 제약조건을 포함한 테이블 생성

CREATE TABLE student (

&#x20;   no INTEGER NOT NULL,

&#x20;   name VARCHAR(30),

&#x20;   birth\_date DATE,

&#x20;   CONSTRAINT pk\_student PRIMARY KEY (no)

);

2\. ALTER TABLE을 통한 구조 변경

ALTER TABLE student ADD age INTEGER; -- 새로운 열 추가

ALTER TABLE student MODIFY name VARCHAR(50); -- 기존 열의 데이터 크기(속성) 변경

ALTER TABLE student DROP age; -- 열 삭제



3\. 데이터 및 테이블 삭제

DELETE FROM student WHERE no = 1; -- 조건에 맞는 특정 데이터(행)만 삭제

TRUNCATE TABLE student; -- 조건 지정 없이 테이블 내부의 '모든 데이터'를 삭제

DROP TABLE student;
